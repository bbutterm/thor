#!/usr/bin/env python3
"""Derive an instrumented, fixed-appearance A/B harness from the preserved stress script."""
from pathlib import Path
import hashlib

source = Path('/workspace/thor-artifacts/room-transition-gl/one_room_uniform_stress.gd').read_text()

def replace_once(old, new):
    global source
    assert source.count(old) == 1, (old, source.count(old))
    source = source.replace(old, new, 1)

replace_once('OneRoomUniformStressRunner', 'OneRoomUniformTimingRunner')
replace_once('var count: int = 20\n', '''var count: int = 20
var spawn_start_frame: int = 0
var spawn_elapsed_ms: float = 0.0
var first_hit_call_ms: float = -1.0
var before_hit_frame: int = 0
var manifest: Array = []
var flash_material_count: int = 0
''')
replace_once('''\tprint("ONE_ROOM_PHASE spawn frame=%d" % Engine.get_process_frames())
\tseed(9017)
\tfor i in count:
''', '''\tprint("ONE_ROOM_PHASE spawn frame=%d" % Engine.get_process_frames())
\tseed(9017)
\t# Measurements are all-frame intervals; disable synchronous hitch-file writes in BOTH arms.
\tPerf.frames.clear()
\tPerf.enabled_logging = false
\tPerf.record_all = true
\tspawn_start_frame = Engine.get_process_frames()
\tvar start_us := Time.get_ticks_usec()
\tfor i in count:
''')
replace_once('''\t\tvar enemy := game.room.spawn_extra("grunt" if i % 4 == 0 else "fodder", pos)
\t\tif enemy:
''', '''\t\t# Same operations as spawn_extra, except appearance is fixed before _ready.
\t\tvar kind := "grunt" if i % 4 == 0 else "fodder"
\t\tvar enemy := game.room._make_enemy(kind)
\t\tenemy.randomize_look = false
\t\tenemy.position = game.room.to_local(pos) + Vector3(0, 0.05, 0)
\t\tif game.room._player:
\t\t\tvar to: Vector3 = game.room._player.global_position - pos
\t\t\tenemy.rotation.y = atan2(to.x, to.z) - game.room.global_rotation.y
\t\tgame.room.add_child(enemy)
\t\tgame.room._register(enemy)
\t\tif enemy:
''')
replace_once('''\t\t\tmob.append(enemy)
\tawait _frames(45)
\t_snapshot("before_hits")
''', '''\t\t\tmob.append(enemy)
\tspawn_elapsed_ms = float(Time.get_ticks_usec() - start_us) / 1000.0
\tawait _frames(45)
\tvar materials := {}
\tfor enemy in mob:
\t\tvar parts: Array = []
\t\tfor gi in enemy._flash_meshes:
\t\t\tvar mesh := gi as MeshInstance3D
\t\t\tparts.append({"name": str(gi.name), "mesh": mesh.mesh.resource_path if mesh and mesh.mesh else ""})
\t\tmanifest.append({"kind": int(enemy.kind), "parts": parts})
\t\tvar flash_material: Material
\t\tfor property in enemy.get_property_list():
\t\t\tif property.name == "_flash_material_gl":
\t\t\t\tflash_material = enemy.get("_flash_material_gl") as Material
\t\t\t\tbreak
\t\tif flash_material == null:
\t\t\tflash_material = Enemy.FLASH_MATERIAL
\t\tmaterials[flash_material.get_instance_id()] = true
\tflash_material_count = materials.size()
\t# Keep manifest inspection out of the first-hit frame.
\tawait _frames(2)
\t_snapshot("before_hits")
\tbefore_hit_frame = Engine.get_process_frames()
''')
replace_once('''\t\t\t\tenemy.take_hit(1.0, Vector3.ZERO, "ranged")
\t\t\t\thits += 1
''', '''\t\t\t\tif hits == 0:
\t\t\t\t\tvar hit_start_us := Time.get_ticks_usec()
\t\t\t\t\tenemy.take_hit(1.0, Vector3.ZERO, "ranged")
\t\t\t\t\tfirst_hit_call_ms = float(Time.get_ticks_usec() - hit_start_us) / 1000.0
\t\t\t\telse:
\t\t\t\t\tenemy.take_hit(1.0, Vector3.ZERO, "ranged")
\t\t\t\thits += 1
''')
replace_once('''\t_check(hits == count * 5, "all scripted ranged hits applied")
''', '''\tPerf.record_all = false
\tPerf.enabled_logging = true
\tvar report := {"spawn_start_frame": spawn_start_frame, "spawn_elapsed_ms": spawn_elapsed_ms,
\t\t"before_hit_frame": before_hit_frame, "first_hit_call_ms": first_hit_call_ms,
\t\t"flash_material_count": flash_material_count, "manifest": manifest, "frames": Perf.frames}
\tvar report_file := FileAccess.open("user://enemy_flash_timing.json", FileAccess.WRITE)
\tif report_file:
\t\treport_file.store_string(JSON.stringify(report))
\t\treport_file.close()
\telse:
\t\t_check(false, "cannot write all-frame report")
\tprint("ONE_ROOM_TIMING spawn_frame=%d spawn_sync_ms=%.3f hit_sync_ms=%.3f flash_materials=%d recorded_frames=%d" % [
\t\tspawn_start_frame, spawn_elapsed_ms, first_hit_call_ms, flash_material_count, Perf.frames.size()])
\t_check(hits == count * 5, "all scripted ranged hits applied")
''')

scene = '''[gd_scene load_steps=2 format=3]

[ext_resource type="Script" path="res://tests/one_room_uniform_timing.gd" id="1"]

[node name="OneRoomUniformTiming" type="Node"]
script = ExtResource("1")
'''
for project in (Path('/workspace/thor-gl-enemy-flash-baseline'), Path('/workspace/thor-gl-enemy-flash')):
    (project / 'tests/one_room_uniform_timing.gd').write_text(source)
    (project / 'tests/one_room_uniform_timing.tscn').write_text(scene)
print('stress script SHA256:', hashlib.sha256(Path('/workspace/thor-artifacts/room-transition-gl/one_room_uniform_stress.gd').read_bytes()).hexdigest())
print('timing script SHA256:', hashlib.sha256(source.encode()).hexdigest())
print('timing scene SHA256:', hashlib.sha256(scene.encode()).hexdigest())
