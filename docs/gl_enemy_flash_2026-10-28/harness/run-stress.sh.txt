#!/usr/bin/env bash
set -u
set -o pipefail
project=/workspace/thor-gl-enemy-flash
out=/workspace/thor-artifacts/gl-enemy-flash-prototype
export LP_NUM_THREADS=4
export LIBGL_ALWAYS_SOFTWARE=1
export THOR_ROOM_PROBE_SYNC=0
ulimit -c 0
ulimit -f 48828
cp /workspace/thor-artifacts/room-transition-gl/xorg-one-room.conf "$out/xorg-stress.conf"
display_num=
for n in $(seq 150 199); do
    if [[ ! -e "/tmp/.X11-unix/X$n" && ! -e "/tmp/.X$n-lock" ]]; then
        display_num=$n
        break
    fi
done
if [[ -z "$display_num" ]]; then printf 'No unused X display\n' >&2; exit 2; fi
/usr/lib/xorg/Xorg ":$display_num" -noreset -nolisten tcp -config "$out/xorg-stress.conf" -logfile "$out/xorg-stress.log" > "$out/xorg-stress-stdout.log" 2>&1 &
xorg_pid=$!
cleanup() {
    kill "$xorg_pid" 2>/dev/null || true
    wait "$xorg_pid" 2>/dev/null || true
}
trap cleanup EXIT
export DISPLAY=":$display_num"
ready=0
for ((i=0; i<50; i++)); do
    if xdpyinfo >/dev/null 2>&1; then ready=1; break; fi
    if ! kill -0 "$xorg_pid" 2>/dev/null; then break; fi
    sleep 0.2
done
if [[ "$ready" != 1 ]]; then printf 'Xorg failed to start\n' >&2; exit 2; fi
printf 'display=%s software=%s threads=%s sync=%s\n' "$DISPLAY" "$LIBGL_ALWAYS_SOFTWARE" "$LP_NUM_THREADS" "$THOR_ROOM_PROBE_SYNC" > "$out/stress-metadata.txt"
sha256sum "$project/scenes/game.gd" "$project/tests/one_room_uniform_stress.gd" "$project/tests/one_room_uniform_stress.tscn" >> "$out/stress-metadata.txt"
for label in N20_C1 N20_C2 N44_D1 N44_D2; do
    if [[ "$label" == N20* ]]; then count=20; else count=44; fi
    mkdir -p "$out/userdata/$label"
    printf 'START %s %s\n' "$label" "$(date -u +%FT%TZ)" | tee -a "$out/stress-metadata.txt"
    if XDG_DATA_HOME="$out/userdata/$label" timeout --signal=TERM --kill-after=10s 240 stdbuf -oL -eL godot --path "$project" --fixed-fps 60 --audio-driver Dummy --rendering-method gl_compatibility --resolution 320x180 res://tests/one_room_uniform_stress.tscn -- "$count" > "$out/$label.log" 2>&1; then
        rc=0
    else
        rc=$?
    fi
    printf 'END %s rc=%d %s\n' "$label" "$rc" "$(date -u +%FT%TZ)" | tee -a "$out/stress-metadata.txt"
    rg 'OpenGL API|ONE_ROOM_(BEGIN|CHECK|END)|Too many instances|SCRIPT ERROR|Parse Error|ERROR:' "$out/$label.log" | head -25 || true
    if [[ "$rc" != 0 ]] || ! rg -q 'ONE_ROOM_END success=true' "$out/$label.log" || rg -q 'SCRIPT ERROR|Parse Error' "$out/$label.log"; then
        printf 'Run %s needs investigation; full log preserved\n' "$label" >&2
        exit 1
    fi
done
sha256sum "$project/scenes/game.gd" "$project/tests/one_room_uniform_stress.gd" "$project/tests/one_room_uniform_stress.tscn" >> "$out/stress-metadata.txt"
